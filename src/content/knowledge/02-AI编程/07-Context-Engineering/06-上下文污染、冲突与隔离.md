# 四、最小实践：在预算内选择上下文


```python
from dataclasses import dataclass


@dataclass(frozen=True)
class ContextItem:
    """保存一个候选上下文块的来源、Token、相关性和新鲜度。"""

    source: str  # 保存可回链的文件或工具来源。
    tokens: int  # 保存使用目标 tokenizer 预估的 Token 数。
    relevance: float  # 保存针对当前任务的相关性分数。
    freshness: float  # 保存仓库版本或时间新鲜度分数。


def select_context(items: list[ContextItem], token_budget: int) -> list[ContextItem]:
    """在 Token 预算内按相关性和新鲜度选择上下文。"""
    # 保存已经装入上下文的 Token 数。
    used_tokens = 0
    # 保存最终选中的上下文块。
    selected_items: list[ContextItem] = []
    for item in sorted(items, key=lambda value: value.relevance * value.freshness, reverse=True):
        if used_tokens + item.tokens > token_budget:
            continue
        selected_items.append(item)
        used_tokens += item.tokens
    return selected_items
```

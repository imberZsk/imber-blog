```python
from pathlib import Path


def resolve_workspace_path(workspace: Path, candidate: str) -> Path:
    """解析并限制候选路径；workspace 是授权根目录，candidate 是外部输入。"""
    # 保存规范化后的授权根目录。
    allowed_root = workspace.resolve()
    # 保存规范化后的候选路径。
    resolved_path = (allowed_root / candidate).resolve()
    if resolved_path != allowed_root and allowed_root not in resolved_path.parents:
        raise PermissionError("path escapes workspace")
    return resolved_path


# 保存演示使用的隔离工作区。
demo_workspace = Path("sandbox")
print(resolve_workspace_path(demo_workspace, "src/app.py"))
try:
    resolve_workspace_path(demo_workspace, "../secret.txt")
except PermissionError as error:
    print(error)
```
